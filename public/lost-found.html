สคริปต์ Google Form Webhook

นำโค้ด Apps Script ไปใส่ใน Google Form เพื่อให้ส่งคำร้องแจ้งของหายเข้าสู่ระบบอัตโนมัติ 

function onFormSubmit(e) {
  // สคริปต์ส่งคำร้องแจ้งของหายจาก Google Form เข้าฐานสารวัตรนักเรียน
  var WEBHOOK_URL = "http://localhost:10001/api/lost-found/claims";
  var responses = e.response.getItemResponses();
  var payload = {
    category: responses[0] ? responses[0].getResponse() : "อื่นๆ",
    item_type: responses[1] ? responses[1].getResponse() : "สิ่งของไม่ระบุชื่อ",
    characteristics: responses[2] ? responses[2].getResponse() : "",
    lost_location: responses[3] ? responses[3].getResponse() : "ไม่ระบุสถานที่",
    lost_date: new Date().toISOString().slice(0, 10),
    claimant_name: responses[4] ? responses[4].getResponse() : "นักเรียน",
    claimant_class: responses[5] ? responses[5].getResponse() : "",
    claimant_contact: responses[6] ? responses[6].getResponse() : ""
  };
  UrlFetchApp.fetch(WEBHOOK_URL, {
    method: "post",
    contentType: "application/json",
    payload: JSON.stringify(payload)
  });
}

สถาปัตยกรรมข้อมูล Supabase & คำสั่งสร้างตาราง

คำสั่ง SQL สำหรับรันใน Supabase Dashboard > SQL Editor เพื่อสร้างตาราง lost_items, lost_claims, และ lost_categories 

-- 1. ตารางคลังสิ่งของที่พบเจอ (Found Items Repository)
CREATE TABLE IF NOT EXISTS public.lost_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    item_code VARCHAR(50) UNIQUE NOT NULL,
    title VARCHAR(200) NOT NULL,
    category VARCHAR(100) NOT NULL,
    characteristics TEXT NOT NULL,
    secret_traits TEXT,
    image_url TEXT,
    location_found VARCHAR(200) NOT NULL,
    found_date DATE NOT NULL DEFAULT CURRENT_DATE,
    found_time VARCHAR(50),
    status VARCHAR(50) DEFAULT 'WAITING_OWNER',
    storage_location VARCHAR(200) DEFAULT 'ฐานหลักสารวัตรนักเรียน หน้าห้องกิจการ',
    finder_name VARCHAR(150),
    finder_contact VARCHAR(100),
    claimant_name VARCHAR(150),
    claimant_contact VARCHAR(100),
    delivered_by VARCHAR(150),
    handover_photo_url TEXT,
    delivered_at TIMESTAMPTZ,
    handover_notes TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. ตารางคำร้องแจ้งของหายจาก Google Form (Lost Claims Feed)
CREATE TABLE IF NOT EXISTS public.lost_claims (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    claim_code VARCHAR(50) UNIQUE NOT NULL,
    category VARCHAR(100) NOT NULL,
    item_type VARCHAR(200) NOT NULL,
    characteristics TEXT NOT NULL,
    lost_location VARCHAR(200),
    lost_date DATE DEFAULT CURRENT_DATE,
    claimant_name VARCHAR(150) NOT NULL,
    claimant_class VARCHAR(50),
    claimant_contact VARCHAR(100) NOT NULL,
    status VARCHAR(50) DEFAULT 'PENDING',
    matched_item_id UUID REFERENCES public.lost_items(id) ON DELETE SET NULL,
    resolved_at TIMESTAMPTZ,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- 3. ตารางหมวดหมู่สิ่งของ (Lost Categories)
CREATE TABLE IF NOT EXISTS public.lost_categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    code VARCHAR(50) UNIQUE NOT NULL,
    name VARCHAR(100) UNIQUE NOT NULL,
    icon VARCHAR(20) DEFAULT 'box',
    description TEXT,
    created_at TIMESTAMPTZ DEFAULT NOW(),
    updated_at TIMESTAMPTZ DEFAULT NOW()
);

-- RLS Policies
ALTER TABLE public.lost_items ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read lost_items" ON public.lost_items FOR SELECT USING (true);
CREATE POLICY "Allow write lost_items" ON public.lost_items FOR ALL USING (true);

ALTER TABLE public.lost_claims ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read lost_claims" ON public.lost_claims FOR SELECT USING (true);
CREATE POLICY "Allow write lost_claims" ON public.lost_claims FOR ALL USING (true);

ALTER TABLE public.lost_categories ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Allow public read lost_categories" ON public.lost_categories FOR SELECT USING (true);
CREATE POLICY "Allow write lost_categories" ON public.lost_categories FOR ALL USING (true);
